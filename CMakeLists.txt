cmake_minimum_required(VERSION 3.22)
project(BASSFORGE VERSION 0.5.0 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(BASSFORGE COMPANY_NAME "Bassforge Audio" BUNDLE_ID com.bassforge.synth PLUGIN_MANUFACTURER_CODE BfAu PLUGIN_CODE Bf01 FORMATS VST3 Standalone PRODUCT_NAME "BASSFORGE" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(BASSFORGE PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_features(BASSFORGE PRIVATE cxx_std_17)
target_compile_definitions(BASSFORGE PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(BASSFORGE PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
juce_generate_juce_header(BASSFORGE)
